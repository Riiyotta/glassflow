async () => {
  const sleep = (t) => new Promise(r => setTimeout(r, t));
  for (let y = 0; y < document.body.scrollHeight; y += 600) { window.scrollTo(0, y); await sleep(100); }
  window.scrollTo(0, 0); await sleep(1500);
  const R = (r) => [Math.round(r.x), Math.round(r.y + scrollY), Math.round(r.width), Math.round(r.height)].join(',');
  const out = [];
  for (const el of document.querySelectorAll('body *')) {
    const cs = getComputedStyle(el);
    if (cs.display === 'none' || cs.visibility === 'hidden') continue;
    const r = el.getBoundingClientRect();
    if (r.width === 0 || r.height === 0) continue;
    const own = [...el.childNodes].filter(n => n.nodeType === 3).map(n => n.textContent).join('').trim();
    let op = 1; for (let e = el; e; e = e.parentElement) op *= parseFloat(getComputedStyle(e).opacity);
    if (own && op > 0.01) out.push('T ' + own.slice(0, 28) + ' | ' + R(r) + ' | ' + cs.fontSize + '/' + cs.lineHeight);
    const bg = cs.backgroundColor;
    if (bg !== 'rgba(0, 0, 0, 0)' && r.width > 20 && r.height > 20 && el.tagName !== 'HTML' && el.tagName !== 'BODY') out.push('B ' + el.tagName + ' ' + R(r) + ' ' + bg + ' pad=' + cs.padding);
  }
  return 'H=' + document.documentElement.scrollHeight + '\n' + out.join('\n');
}
